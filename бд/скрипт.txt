-- ============================================================
-- 0. Расширения и схема
-- ============================================================
CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE SCHEMA IF NOT EXISTS dev_team;

-- ============================================================
-- 1. Enum-типы (остались только project_status и user_role)
-- ============================================================
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_type t
                   JOIN pg_namespace n ON n.oid = t.typnamespace
                   WHERE t.typname = 'user_role' AND n.nspname = 'dev_team') THEN
        CREATE TYPE dev_team.user_role AS ENUM ('customer', 'manager', 'developer');
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_type t
                   JOIN pg_namespace n ON n.oid = t.typnamespace
                   WHERE t.typname = 'project_status' AND n.nspname = 'dev_team') THEN
        CREATE TYPE dev_team.project_status AS ENUM
            ('draft', 'in_progress', 'pending_approval', 'completed', 'closed', 'frozen');
    END IF;
END$$;

-- ============================================================
-- 2. Таблицы
-- ============================================================

-- 2.1. company
CREATE TABLE IF NOT EXISTS dev_team.company (
    id_company  SERIAL PRIMARY KEY,
    name        VARCHAR(255) NOT NULL
);

-- 2.2. team
CREATE TABLE IF NOT EXISTS dev_team.team (
    id_team     SERIAL PRIMARY KEY,
    id_company  INTEGER NOT NULL,
    name        VARCHAR(255) NOT NULL,
    description TEXT
);

-- 2.3. users
CREATE TABLE IF NOT EXISTS dev_team.users (
    id_user           SERIAL PRIMARY KEY,
    id_company        INTEGER,
    id_team           INTEGER,
    first_name        VARCHAR(100) NOT NULL,
    last_name         VARCHAR(100) NOT NULL,
    role              dev_team.user_role NOT NULL,
    login             VARCHAR(100) NOT NULL,
    password_hash     VARCHAR(255) NOT NULL,
    short_description TEXT
);

-- 2.4. invoice (до project — project.id_invoice ссылается сюда)
CREATE TABLE IF NOT EXISTS dev_team.invoice (
    id_invoice  SERIAL PRIMARY KEY,
    name        VARCHAR(255) NOT NULL,
    cost        NUMERIC(15,2) NOT NULL DEFAULT 0,
    start_date  DATE,
    end_date    DATE
);

-- 2.5. project (1:1 с invoice через id_invoice UNIQUE)
CREATE TABLE IF NOT EXISTS dev_team.project (
    id_project  SERIAL PRIMARY KEY,
    id_customer INTEGER NOT NULL,
    id_team     INTEGER,
    id_invoice  INTEGER,
    name        VARCHAR(255) NOT NULL,
    description TEXT,
    status      dev_team.project_status NOT NULL DEFAULT 'draft'
);

-- 2.6. technical_specification
CREATE TABLE IF NOT EXISTS dev_team.technical_specification (
    id_spec     SERIAL PRIMARY KEY,
    id_project  INTEGER NOT NULL,
    name        VARCHAR(255) NOT NULL,
    deadline    DATE,
    description TEXT
);

-- 2.7. specification_item
CREATE TABLE IF NOT EXISTS dev_team.specification_item (
    id_item     SERIAL PRIMARY KEY,
    id_spec     INTEGER NOT NULL,
    name        VARCHAR(255) NOT NULL,
    description TEXT
);

-- 2.8. task_list
CREATE TABLE IF NOT EXISTS dev_team.task_list (
    id_task_list       SERIAL PRIMARY KEY,
    id_invoice         INTEGER NOT NULL,
    name               VARCHAR(255) NOT NULL,
    completion_percent NUMERIC(5,2) NOT NULL DEFAULT 0
);

-- 2.9. task (status — varchar, не enum)
CREATE TABLE IF NOT EXISTS dev_team.task (
    id_task       SERIAL PRIMARY KEY,
    id_task_list  INTEGER NOT NULL,
    id_developer  INTEGER,
    description   TEXT,
    status        VARCHAR(50) NOT NULL DEFAULT 'new',
    planned_hours NUMERIC(7,2),
    actual_hours  NUMERIC(7,2),
    cost          NUMERIC(15,2)
);

-- 2.10. change_request (id_spec_item nullable, status — varchar)
CREATE TABLE IF NOT EXISTS dev_team.change_request (
    id_change     SERIAL PRIMARY KEY,
    id_spec_item  INTEGER,
    id_task       INTEGER,
    id_invoice    INTEGER,
    id_creator    INTEGER NOT NULL,
    id_approver   INTEGER,
    name          VARCHAR(255) NOT NULL,
    description   TEXT,
    status        VARCHAR(50) NOT NULL DEFAULT 'created',
    created_at    TIMESTAMP WITHOUT TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 6.1. Пересчёт completion_percent для одного списка задач
CREATE OR REPLACE FUNCTION dev_team.fn_recalc_task_list_completion_for(p_task_list_id integer)
RETURNS void
LANGUAGE plpgsql
AS $function$
DECLARE
    v_total_weight  NUMERIC(15, 2);
    v_done_weight   NUMERIC(15, 2);
    v_percent       NUMERIC(5, 2);
BEGIN
    SELECT COALESCE(SUM(COALESCE(planned_hours, 1)), 0)
      INTO v_total_weight
      FROM dev_team.task
     WHERE id_task_list = p_task_list_id;

    SELECT COALESCE(SUM(COALESCE(planned_hours, 1)), 0)
      INTO v_done_weight
      FROM dev_team.task
     WHERE id_task_list = p_task_list_id
       AND status = 'done';

    IF v_total_weight > 0 THEN
        v_percent := ROUND((v_done_weight / v_total_weight) * 100, 2);
    ELSE
        v_percent := 0;
    END IF;

    UPDATE dev_team.task_list
       SET completion_percent = v_percent
     WHERE id_task_list = p_task_list_id;
END;
$function$;

-- 6.2. Триггерная функция пересчёта
CREATE OR REPLACE FUNCTION dev_team.fn_recalc_task_list_completion()
RETURNS trigger
LANGUAGE plpgsql
AS $function$
DECLARE
    v_task_list_id  INTEGER;
BEGIN
    IF TG_OP = 'DELETE' THEN
        v_task_list_id := OLD.id_task_list;
    ELSE
        v_task_list_id := NEW.id_task_list;
    END IF;

    PERFORM dev_team.fn_recalc_task_list_completion_for(v_task_list_id);

    IF TG_OP = 'UPDATE' AND OLD.id_task_list <> NEW.id_task_list THEN
        PERFORM dev_team.fn_recalc_task_list_completion_for(OLD.id_task_list);
    END IF;

    RETURN NULL;
END;
$function$;

-- 6.3. Триггер на task
DROP TRIGGER IF EXISTS trg_task_recalc_completion ON dev_team.task;
CREATE TRIGGER trg_task_recalc_completion
    AFTER INSERT OR UPDATE OR DELETE ON dev_team.task
    FOR EACH ROW
    EXECUTE FUNCTION dev_team.fn_recalc_task_list_completion();

-- 6.4. Guard-функция для change_request
CREATE OR REPLACE FUNCTION dev_team.fn_change_request_guard()
RETURNS trigger
LANGUAGE plpgsql
AS $function$
BEGIN
    IF TG_OP = 'UPDATE' AND NEW.created_at <> OLD.created_at THEN
        NEW.created_at := OLD.created_at;
    END IF;

    IF TG_OP = 'UPDATE'
       AND OLD.status = 'approved'
       AND NEW.status <> 'approved' THEN
        RAISE EXCEPTION 'Нельзя изменить статус одобренного изменения (id=%)', OLD.id_change;
    END IF;

    IF TG_OP = 'UPDATE'
       AND NEW.status = 'approved'
       AND NEW.id_approver IS NULL THEN
        RAISE EXCEPTION 'Для одобрения изменения (id=%) требуется id_approver', NEW.id_change;
    END IF;

    RETURN NEW;
END;
$function$;

-- 6.5. Триггер на change_request
DROP TRIGGER IF EXISTS trg_change_request_guard ON dev_team.change_request;
CREATE TRIGGER trg_change_request_guard
    BEFORE UPDATE ON dev_team.change_request
    FOR EACH ROW
    EXECUTE FUNCTION dev_team.fn_change_request_guard();
